type smpProfileWorker struct {
	Nodes   uint64            `json:"nodes"`
	Depth   int               `json:"depth"`
	Move    string            `json:"move"`
	Samples map[uint64]uint64 `json:"samples,omitempty"`
}

// Replaced only in the diagnostic overlay; the native build has no hot-path edits.
func smpProfileSamples(info *SearchInfo) map[uint64]uint64 { return nil }

func smpProfileInt(t *testing.T, name string) int {
	t.Helper()
	v, err := strconv.Atoi(os.Getenv(name))
	if err != nil {
		t.Fatal(name, err)
	}
	return v
}

func TestSMPProfile(t *testing.T) {
	if runtime.GOMAXPROCS(0) != 2 {
		t.Fatal("requires GOMAXPROCS=2")
	}
	model, err := rodentv12eval.LoadV12Default(os.Getenv("RODENT_V12_DEFAULT_MODEL"))
	if err != nil {
		t.Fatal(err)
	}
	metadata, err := model.Metadata()
	if err != nil {
		t.Fatal(err)
	}
	width := smpProfileInt(t, "SMP_WIDTH")
	ms := smpProfileInt(t, "SMP_MS")
	nodes := smpProfileInt(t, "SMP_NODES")
	searcher, err := NewSearchEngineWithHash(128)
	if err != nil {
		t.Fatal(err)
	}
	if err := searcher.SelectRodentV12DefaultEvaluator(model); err != nil {
		t.Fatal(err)
	}
	if err := searcher.ConfigureThreads(width); err != nil {
		t.Fatal(err)
	}
	pos, err := ParseFEN(os.Getenv("SMP_FEN"))
	if err != nil {
		t.Fatal(err)
	}
	pos.Hash()
	before := snapshotPVPosition(pos)
	workers := make([]smpProfileWorker, width)
	var mu sync.Mutex
	snapshot := func(index int, info *SearchInfo) {
		mu.Lock()
		defer mu.Unlock()
		workers[index] = smpProfileWorker{info.Nodes, info.Depth, info.BestMove.ToString(), smpProfileSamples(info)}
	}
	searcher.smpHooks = &smpSearchHooks{workerFinished: snapshot}
	if nodes > 0 {
		searcher.SetMaxNodes(uint64(nodes))
	}
	tm := NewTimeManager()
	if err := tm.SetMoveOverhead(0); err != nil {
		t.Fatal(err)
	}
	profile := os.Getenv("SMP_PROFILES") == "1"
	out := os.Getenv("SMP_OUT")
	if profile {
		f, err := os.Create(filepath.Join(out, "cpu.pprof"))
		if err != nil {
			t.Fatal(err)
		}
		if err := pprof.StartCPUProfile(f); err != nil {
			t.Fatal(err)
		}
		defer f.Close()
		runtime.SetMutexProfileFraction(1)
		runtime.SetBlockProfileRate(1)
	}
	start := time.Now()
	var info *SearchInfo
	pprof.Do(context.Background(), pprof.Labels("scope", "search"), func(context.Context) {
		if nodes > 0 {
			info = searcher.SearchIterativeDeepening(pos, MaximumDepth-1, nil)
		} else {
			tm.SetTimeControlAt(SearchParams{MoveTime: ms}, pos.Turn() == White, start)
			info = searcher.SearchIterativeDeepening(pos, MaximumDepth-1, tm)
		}
	})
	elapsed := time.Since(start).Seconds()
	if profile {
		pprof.StopCPUProfile()
		runtime.SetMutexProfileFraction(0)
		runtime.SetBlockProfileRate(0)
		for _, name := range []string{"mutex", "block"} {
			f, err := os.Create(filepath.Join(out, name+".pprof"))
			if err != nil {
				t.Fatal(err)
			}
			if err := pprof.Lookup(name).WriteTo(f, 0); err != nil {
				t.Fatal(err)
			}
			if err := f.Close(); err != nil {
				t.Fatal(err)
			}
		}
	}
	if info.EffectiveThreads == 1 {
		snapshot(0, info)
	}
	if info.Nodes == 0 || info.BestMove == EmptyMove {
		t.Fatal("empty search")
	}
	if nodes > 0 {
		if info.EffectiveThreads != 1 || info.Nodes < uint64(nodes) {
			t.Fatal("invalid one-worker control")
		}
	} else if info.EffectiveThreads != width || elapsed < float64(ms)*0.9/1000 {
		t.Fatal("invalid fixed-time width/duration", info.EffectiveThreads, elapsed)
	}
	if !reflectSMPProfileRoot(pos, before) {
		t.Fatal("root not restored")
	}
	for _, worker := range searcher.workerSnapshot().workers {
		if worker.evaluator.nnueDepth() != 0 {
			t.Fatal("context not restored")
		}
	}
	row := map[string]any{
		"model_sha256": metadata.SHA256, "model_bytes": metadata.Bytes,
		"width": width, "effective_threads": info.EffectiveThreads, "gomaxprocs": runtime.GOMAXPROCS(0),
		"seconds": elapsed, "nodes": info.Nodes, "nps": float64(info.Nodes) / elapsed,
		"depth": info.Depth, "move": info.BestMove.ToString(), "score": info.BestScore,
		"tt_probes": info.TTProbes, "tt_hits": info.TTHits, "tt_cutoffs": info.TTCutoffs,
		"qtt_probes": info.QTTProbes, "qtt_hits": info.QTTHits, "qtt_cutoffs": info.QTTCutoffs,
		"workers": workers,
	}
	data, err := json.MarshalIndent(row, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(filepath.Join(out, "result.json"), append(data, '\n'), 0644); err != nil {
		t.Fatal(err)
	}
}

func reflectSMPProfileRoot(pos *Position, before pvPositionSnapshot) bool {
	// Reuse the accepted package's complete PV-position snapshot contract.
	return reflect.DeepEqual(snapshotPVPosition(pos), before)
}
