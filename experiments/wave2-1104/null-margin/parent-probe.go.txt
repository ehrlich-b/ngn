package main

import (
 "encoding/json"
 "os"
 "github.com/ehrlich-b/ngn/engine"
)
func must(err error) { if err != nil { panic(err) } }
func main() {
 data, err := os.ReadFile(os.Args[1]); must(err)
 var fens []string; must(json.Unmarshal(data, &fens))
 net, err := engine.LoadDefaultNet(); must(err)
 engine.W2NullVerify = 1
 enc := json.NewEncoder(os.Stdout)
 for _, fen := range fens {
  e, err := engine.NewSearchEngineWithHash(1); must(err)
  must(e.SelectNGNN1Evaluator(net))
  p, err := engine.ParseFEN(fen); must(err)
  info := e.SearchIterativeDeepening(p, 4, nil)
  pv := []string{}
  for _, move := range e.ExtractPVFromTT(p, info.BestMove, 4) { pv = append(pv, move.ToString()) }
  must(enc.Encode(map[string]interface{}{"fen":fen,"depth":info.Depth,"nodes":info.Nodes,"qnodes":info.QNodes,"score":info.BestScore,"best":info.BestMove.ToString(),"pv":pv,"null_tries":info.NullMoveTries,"null_cutoffs":info.NullMoveCutoffs,"stopped":info.Stopped,"net_sha256":engine.DefaultNetSHA256}))
 }
}
