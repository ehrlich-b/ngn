package engine

import (
	"encoding/json"
	"math"
	"os"
	"testing"
)

// This is a diagnostic fixture, never a playing-source change. The TT stores
// the full 64-bit key; Hash64 uses the lowest20 bits to select four-way buckets.
func TestCampaignKeyDiagnostics(t *testing.T) {
	keys := []uint64{whiteSideKey}
	keys = append(keys, castlingKeys[:]...)
	keys = append(keys, epTargetKeys[:]...)
	for square := range occupantKeys {
		keys = append(keys, occupantKeys[square][1:]...)
	}
	seen := map[uint64]bool{}
	var counts [64]int
	duplicates, zeros := 0, 0
	for _, key := range keys {
		if seen[key] {
			duplicates++
		}
		seen[key] = true
		if key == 0 {
			zeros++
		}
		for bit := range counts {
			counts[bit] += int(key >> bit & 1)
		}
	}
	maxZ := 0.0
	for _, count := range counts {
		z := math.Abs(float64(count)-float64(len(keys))/2) / math.Sqrt(float64(len(keys))/4)
		maxZ = math.Max(maxZ, z)
	}
	if duplicates != 0 || zeros != 0 || maxZ > 4.5 {
		t.Errorf("key-table defect: duplicates=%d zeros=%d worst bit z=%.3f", duplicates, zeros, maxZ)
	}
	net, err := LoadNGNN1(os.Getenv("NGN_KEY_NET"))
	if err != nil {
		t.Fatal(err)
	}
	fens := []string{
		"rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1",
		"r1bq1rk1/pp2bppp/2n2n2/2pp4/3P4/2N1PN2/PPQ1BPPP/R1B2RK1 w - - 0 10",
		"r3k2r/p1ppqpb1/bn2pnp1/3PN3/1p2P3/2N2Q1p/PPPBBPPP/R3K2R w KQkq - 0 1",
		"r4rk1/1pp1qppp/p1np1n2/2b1p1B1/2B1P1b1/P1NP1N2/1PP1QPPP/R4RK1 w - - 0 10",
		"8/2p5/3p4/KP5r/1R3p1k/8/4P1P1/8 w - - 0 1",
		"8/5k2/2p2p2/3p4/3P1P2/2P2K2/8/8 w - - 0 1",
		"6k1/5pp1/7p/3q4/8/3Q2P1/5P1P/6K1 w - - 0 1",
		"4k3/8/8/3pP3/8/8/8/4K3 w - d6 0 1",
	}
	rows := []map[string]any{}
	for _, fen := range fens {
		p, err := ParseFEN(fen)
		if err != nil {
			t.Fatal(err)
		}
		e, err := NewSearchEngineWithHash(64)
		if err != nil {
			t.Fatal(err)
		}
		if err := e.SelectNGNN1Evaluator(net); err != nil {
			t.Fatal(err)
		}
		ngnKeyStats.Probes, ngnKeyStats.Hits = 0, 0
		ngnKeyStats.IndexCollisions, ngnKeyStats.Evictions = 0, 0
		info := e.SearchIterativeDeepening(p, 8, nil)
		rows = append(rows, map[string]any{"fen": fen, "depth": info.Depth,
			"nodes": info.Nodes, "score": info.BestScore, "best": info.BestMove.ToString(),
			"probes": ngnKeyStats.Probes, "hits": ngnKeyStats.Hits,
			"occupied_index_collisions": ngnKeyStats.IndexCollisions,
			"evictions": ngnKeyStats.Evictions})
	}
	data, err := json.MarshalIndent(map[string]any{"key_count": len(keys),
		"duplicates": duplicates, "zero_keys": zeros, "one_counts_by_bit": counts,
		"worst_bit_z": maxZ, "bit_balance_ceiling_z": 4.5,
		"hash_mb": 64, "index_bits": 20, "stored_check_bits": 64,
		"rows": rows}, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(os.Getenv("NGN_KEY_OUT"), append(data, '\n'), 0600); err != nil {
		t.Fatal(err)
	}
}
